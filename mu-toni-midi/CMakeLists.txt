# mu-Toni MIDI — mu-Toni's arpeggiator as a MIDI effect (VST3 + CLAP, no audio, no Standalone).
#
# The arp's clock and note scan are mu-Toni's own (mu-toni/Source/Sequencer/ArpRunner.h, shared with the synth); this product
# drives a stand-in voice that writes MIDI notes. It compiles the arp parameter rows from mu-toni (addArpRows), a few mu-core
# sources behind them, and nothing else of mu-Toni. See CLAUDE.md.

# ── mu-Toni MIDI — VST3 + CLAP (+ AU on macOS) ──────────────────────────────
juce_add_plugin(mu-toni-midi
    COMPANY_NAME "Transwarp Development Project"
    BUNDLE_ID "com.transwarp.mu-toni-midi"
    PLUGIN_MANUFACTURER_CODE TDP1
    PLUGIN_CODE MTnM
    FORMATS VST3 ${MUFAMILY_AU_FORMAT}   # AU on macOS only (family rule); no Standalone
    PRODUCT_NAME "mu-Toni MIDI"
    PLUGIN_NAME  "μ-Toni MIDI"
    IS_SYNTH FALSE
    NEEDS_MIDI_INPUT TRUE
    NEEDS_MIDI_OUTPUT TRUE
    IS_MIDI_EFFECT TRUE
    EDITOR_WANTS_KEYBOARD_FOCUS FALSE
    VERSION         "${MUFAMILY_PRE_PLUGIN_VERSION}"   # pre-release line, as mu-Toni
    COMPANY_COPYRIGHT "${MUFAMILY_COPYRIGHT}"
    ICON_BIG   "${CMAKE_SOURCE_DIR}/mu-toni/resources/icon.png"
    ICON_SMALL "${CMAKE_SOURCE_DIR}/mu-toni/resources/icon.png"
)

# Custom VERSIONINFO with the Greek mu (Windows only), regenerated by the increment target each build.
if(WIN32)
    configure_file(
        "${CMAKE_CURRENT_SOURCE_DIR}/resources/versioninfo.rc.in"
        "${CMAKE_CURRENT_BINARY_DIR}/versioninfo.rc"
        @ONLY
        NEWLINE_STYLE CRLF
    )
    target_compile_definitions(mu-toni-midi        PRIVATE "JUCE_USER_DEFINED_RC_FILE=\"${CMAKE_CURRENT_BINARY_DIR}/versioninfo.rc\"")
    target_compile_definitions(mu-toni-midi_rc_lib PRIVATE "JUCE_USER_DEFINED_RC_FILE=\"${CMAKE_CURRENT_BINARY_DIR}/versioninfo.rc\"")
    add_dependencies(mu-toni-midi_rc_lib increment_build_number)
endif()

# What the arp parameter rows need from mu-Toni and mu-core (the same list mu-toni-layer-tests links).
set(MU_TONI_MIDI_SHARED_SOURCES
    "${CMAKE_SOURCE_DIR}/mu-toni/Source/Persistence/ArpParams.cpp"
    "${CMAKE_SOURCE_DIR}/mu-core/Audio/Wavetable/WavetableBank.cpp"
    "${CMAKE_SOURCE_DIR}/mu-core/Audio/InsertProcessor.cpp"
    "${CMAKE_SOURCE_DIR}/mu-core/Audio/MultiModeFilter.cpp"
)

target_sources(mu-toni-midi
    PRIVATE
        Source/Plugin/PluginProcessor.cpp
        Source/Plugin/PluginEditor.cpp
        Source/Sequencer/MidiArpCore.cpp
        ${MU_TONI_MIDI_SHARED_SOURCES}
)

# This product's Source/ comes first so its Plugin/PluginProcessor.h wins over mu-Toni's of the same path.
target_include_directories(mu-toni-midi
    PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/Source"
        "${CMAKE_SOURCE_DIR}/mu-toni/Source"
    SYSTEM PRIVATE
        "${CMAKE_SOURCE_DIR}/ThirdParty/signalsmith-basics/include"
        "${CMAKE_SOURCE_DIR}/ThirdParty/signalsmith-dsp/include"
)

target_compile_definitions(mu-toni-midi
    PUBLIC
        JUCE_WEB_BROWSER=0
        JUCE_USE_CURL=0
        JUCE_VST3_CAN_REPLACE_VST2=0
        JUCE_USE_MP3AUDIOFORMAT=1
)

target_link_libraries(mu-toni-midi
    PRIVATE
        mu-core
        juce::juce_audio_utils
        juce::juce_dsp
        juce::juce_gui_basics
        juce::juce_gui_extra
        juce::juce_audio_formats
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_lto_flags
        juce::juce_recommended_warning_flags
)

# Family build counter (the shared increment target is defined in mu-clid/CMakeLists.txt).
add_dependencies(mu-toni-midi increment_build_number)

clap_juce_extensions_plugin(TARGET mu-toni-midi
    CLAP_ID "com.transwarp.mu-toni-midi"
    CLAP_FEATURES note-effect)
