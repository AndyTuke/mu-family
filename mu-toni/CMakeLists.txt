# mu-toni — μ-family generative arpeggiator mono-synth (freeware).
#
# The target combines the product's arp + voice engine with the shared platform
# (ProcessorBase + MixerEngine + FXChain + the shared editor shell / sidebar /
# mixer). Shared audio/UI/ProcessorBase come from mu-core via the link.

# ── mu-Toni — Standalone + VST3 + CLAP ──────────────────────────────────────
juce_add_plugin(mu-toni
    COMPANY_NAME "Transwarp Development Project"
    BUNDLE_ID "com.transwarp.mu-toni"
    PLUGIN_MANUFACTURER_CODE TDP1
    PLUGIN_CODE MTni
    FORMATS VST3 Standalone ${MUFAMILY_AU_FORMAT}   # AU on macOS only (family rule)
    PRODUCT_NAME "mu-Toni"
    PLUGIN_NAME  "μ-Toni"
    IS_SYNTH TRUE
    NEEDS_MIDI_INPUT TRUE
    NEEDS_MIDI_OUTPUT FALSE
    IS_MIDI_EFFECT FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS FALSE
    VERSION         "${MUFAMILY_PRE_PLUGIN_VERSION}"   # pre-release 0.1 line
    COMPANY_COPYRIGHT "${MUFAMILY_COPYRIGHT}"
    ICON_BIG   "${CMAKE_CURRENT_SOURCE_DIR}/resources/icon.png"
    ICON_SMALL "${CMAKE_CURRENT_SOURCE_DIR}/resources/icon.png"
)

target_sources(mu-toni
    PRIVATE
        Source/Plugin/PluginProcessor.cpp
        Source/Plugin/PluginProcessor_Preset.cpp
        Source/Plugin/PluginEditor.cpp
        Source/UI/SettingsOverlay.cpp
        Source/UI/ArpContent.cpp
        # Offline-license verifier + online activation live in mu-core (shared); compiled here + monocypher.c.
        "${CMAKE_SOURCE_DIR}/mu-core/License/LicenseManager.cpp"
        "${CMAKE_SOURCE_DIR}/mu-core/License/LemonSqueezyClient.cpp"
        "${CMAKE_SOURCE_DIR}/mu-core/License/OnlineActivation.cpp"
        "${CMAKE_SOURCE_DIR}/ThirdParty/monocypher/monocypher.c"
)

# monocypher is vendored third-party crypto; silence its benign MSVC C4244 at the build level.
if(MSVC)
    set_source_files_properties(
        "${CMAKE_SOURCE_DIR}/ThirdParty/monocypher/monocypher.c"
        PROPERTIES COMPILE_OPTIONS "/wd4244")
endif()

target_include_directories(mu-toni
    PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/Source"
    SYSTEM PRIVATE
        "${CMAKE_SOURCE_DIR}/ThirdParty/signalsmith-basics/include"
        "${CMAKE_SOURCE_DIR}/ThirdParty/signalsmith-dsp/include"
        "${CMAKE_SOURCE_DIR}/ThirdParty/monocypher"   # LicenseManager
)

target_compile_definitions(mu-toni
    PUBLIC
        JUCE_WEB_BROWSER=0
        JUCE_USE_CURL=0
        JUCE_VST3_CAN_REPLACE_VST2=0
        "MU_VERSION_BASE=\"${MUFAMILY_PRE_VERSION_BASE}\""   # pre-release 0.1 line shown in About
)

target_link_libraries(mu-toni
    PRIVATE
        mu-core
        juce::juce_audio_utils
        juce::juce_dsp
        juce::juce_gui_basics
        juce::juce_gui_extra
        juce::juce_audio_formats
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_lto_flags
        juce::juce_recommended_warning_flags
)

# Custom standalone app — shared "Close μ-Toni?" confirmation on quit (mirrors
# mu-clid / mu-tant). Compiled only into the Standalone target.
target_compile_definitions(mu-toni_Standalone PRIVATE JUCE_USE_CUSTOM_PLUGIN_STANDALONE_APP=1)
target_sources(mu-toni_Standalone PRIVATE
    Source/Plugin/StandaloneApp.cpp)

# Hardware controller support (mu-control) in this Standalone only; a no-op unless MUFAMILY_WITH_CONTROL is ON.
mu_control_attach(mu-toni_Standalone)

# CLAP format via clap-juce-extensions (mirrors mu-clid / mu-tant). Creates mu-toni_CLAP.
clap_juce_extensions_plugin(TARGET mu-toni
    CLAP_ID "com.transwarp.mu-toni"
    CLAP_FEATURES instrument synthesizer)

# Family build-counter (shared increment target defined in mu-clid/CMakeLists.txt).
add_dependencies(mu-toni increment_build_number)

# Custom Windows VERSIONINFO with the full four-part family version (JUCE's generated .rc
# can't take x.y.0.N and falls back to 1.0.0). Configured now, then regenerated by the
# increment target each build so file properties match the About panel.
if(WIN32)
    configure_file(
        "${CMAKE_CURRENT_SOURCE_DIR}/resources/versioninfo.rc.in"
        "${CMAKE_CURRENT_BINARY_DIR}/versioninfo.rc"
        @ONLY
        NEWLINE_STYLE CRLF   # RC compiler expects CRLF on Windows
    )
    target_compile_definitions(mu-toni        PRIVATE "JUCE_USER_DEFINED_RC_FILE=\"${CMAKE_CURRENT_BINARY_DIR}/versioninfo.rc\"")
    target_compile_definitions(mu-toni_rc_lib PRIVATE "JUCE_USER_DEFINED_RC_FILE=\"${CMAKE_CURRENT_BINARY_DIR}/versioninfo.rc\"")
    add_dependencies(mu-toni_rc_lib increment_build_number)
endif()

# ── Deploy Release artifacts to the tester drop folder ──────────────────────
# Mirrors mu-tant's deploy pattern. Guarded by MUFAMILY_DEPLOY_TESTERS (OFF by default).
if(MUFAMILY_DEPLOY_TESTERS)
add_custom_command(TARGET mu-toni_Standalone POST_BUILD
    COMMAND "${CMAKE_COMMAND}"
            "-DCONFIG=$<CONFIG>"
            "-DSTANDALONE=$<TARGET_FILE:mu-toni_Standalone>"
            "-DVST3_BUNDLE=${CMAKE_BINARY_DIR}/mu-toni/mu-toni_artefacts/$<CONFIG>/VST3/mu-Toni.vst3"
            "-DCLAP_FILE=${CMAKE_BINARY_DIR}/mu-toni/mu-toni_artefacts/$<CONFIG>/CLAP/mu-Toni.clap"
            "-DDIST_WIN=${MUFAMILY_WIN_DIST}"
            "-DPRODUCT_FILENAME=mu-Toni"
            -P "${CMAKE_SOURCE_DIR}/cmake/DeployRelease.cmake"
    COMMENT "Deploying mu-Toni Release artifacts to OneDrive (Debug skipped)"
    VERBATIM
)
endif()

# ── mu-toni-tests — console UnitTests ───────────────────────────────────────
# Built as part of ALL_BUILD (mirrors mu-tant-tests: small + local, not a tester
# deploy, so building every build is cheap — the speed concession that keeps
# mu-clid-tests EXCLUDE_FROM_ALL doesn't apply here).
include("${CMAKE_SOURCE_DIR}/mu-core/cmake/ProcessorTestSources.cmake")
juce_add_console_app(mu-toni-tests PRODUCT_NAME "mu-toni-tests")

    Source/Tests/ArpRunnerTests.cpp     # T3 — arp step timing (beat grid / MIDI trigger)
    Source/Tests/ProcessorSeamTests.cpp  # the real PluginProcessor, headless (layer ownership)

    Source/Tests/FxTailTests.cpp   # a full-preset commit keeps the FX tail
    Source/Tests/ModArcTests.cpp   # every mod target publishes its live value for the knob arcs
    Source/Tests/StepPatternTests.cpp   # step pattern: rests, Legato tie end, defaults
    Source/Tests/GoldenRenderTests.cpp   # golden renders (tests/golden/mu-toni) and the behaviour pins
    # The real PluginProcessor, built headless (createEditor lives in PluginEditor.cpp, not linked).

    Source/Plugin/PluginProcessor_Preset.cpp
