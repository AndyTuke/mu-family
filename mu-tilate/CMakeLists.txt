# mu-tilate — μ-family loop slicer (planned product, built in stages; docs/mu-tilate/design-buildplan.md).
#
# Built in stages: the layer library and its console tests, then (stage 5) the plugin target below.

# ── mu-tilate-layer — the Phrase layer type as a library (docs/mu-tronic/design.md 2.5) ──
# The product's own Source/ folders: Audio/ Sequencer/ Modulation/ Persistence/ and UI/. INTERFACE, like mu-core,
# so its sources compile in each consumer's target; tests/scripts/check-layer-boundary.ps1 keeps it free of
# Plugin/ and License/.
add_library(mu-tilate-layer INTERFACE)
target_sources(mu-tilate-layer INTERFACE
    "${CMAKE_CURRENT_SOURCE_DIR}/Source/Audio/SlicePlayer.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/Source/Persistence/PhraseParams.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/Source/Persistence/PhraseState.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/Source/Sequencer/SliceList.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/Source/Sequencer/Timeline.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/Source/Sequencer/PhrasePlayback.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/Source/Sequencer/Phrase.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/Source/Sequencer/PhraseAnalysis.cpp"
)
target_include_directories(mu-tilate-layer INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/Source")
# Signalsmith Stretch (MIT, header-only) and its FFT library; SYSTEM so their warnings stay out of ours.
target_include_directories(mu-tilate-layer SYSTEM INTERFACE
    "${CMAKE_SOURCE_DIR}/ThirdParty/signalsmith-stretch"
    "${CMAKE_SOURCE_DIR}/ThirdParty/signalsmith-linear/include")

set(MU_TILATE_LAYER_UI_SOURCES
    "${CMAKE_CURRENT_SOURCE_DIR}/Source/UI/PhraseContent.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/Source/UI/PhraseTimeline.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/Source/UI/SlicePanel.cpp"
)
# Also visible to the sibling directories added after this one (mu-Tronic will host the Phrase type).
set(MU_TILATE_LAYER_UI_SOURCES "${MU_TILATE_LAYER_UI_SOURCES}" PARENT_SCOPE)

# ── mu-Tilate — Standalone + VST3 + CLAP ────────────────────────────────────
juce_add_plugin(mu-tilate
    COMPANY_NAME "Transwarp Development Project"
    BUNDLE_ID "com.transwarp.mu-tilate"
    PLUGIN_MANUFACTURER_CODE TDP1
    PLUGIN_CODE MTil
    FORMATS VST3 Standalone ${MUFAMILY_AU_FORMAT}   # AU on macOS only (family rule)
    PRODUCT_NAME "mu-Tilate"
    PLUGIN_NAME  "μ-Tilate"
    IS_SYNTH TRUE
    NEEDS_MIDI_INPUT TRUE
    NEEDS_MIDI_OUTPUT FALSE
    IS_MIDI_EFFECT FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS FALSE
    VERSION         "${MUFAMILY_PRE_PLUGIN_VERSION}"   # pre-release line
    COMPANY_COPYRIGHT "${MUFAMILY_COPYRIGHT}"
    ICON_BIG   "${CMAKE_CURRENT_SOURCE_DIR}/resources/icon.png"
    ICON_SMALL "${CMAKE_CURRENT_SOURCE_DIR}/resources/icon.png"
)

target_sources(mu-tilate
    PRIVATE
        Source/Plugin/PluginProcessor.cpp
        Source/Plugin/PluginProcessor_Preset.cpp
        Source/Plugin/PluginEditor.cpp
        ${MU_TILATE_LAYER_UI_SOURCES}   # the layer's UI content (mu-tilate-layer)
)

target_include_directories(mu-tilate
    PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/Source"
    SYSTEM PRIVATE
        "${CMAKE_SOURCE_DIR}/ThirdParty/signalsmith-basics/include"
        "${CMAKE_SOURCE_DIR}/ThirdParty/signalsmith-dsp/include"
)

target_compile_definitions(mu-tilate
    PUBLIC
        JUCE_WEB_BROWSER=0
        JUCE_USE_CURL=0
        JUCE_USE_MP3AUDIOFORMAT=1   # the loop chooser offers .mp3
        JUCE_VST3_CAN_REPLACE_VST2=0
        "MU_VERSION_BASE=\"${MUFAMILY_PRE_VERSION_BASE}\""   # pre-release line shown in About
)

target_link_libraries(mu-tilate
    PRIVATE
        mu-tilate-layer
        mu-core
        juce::juce_audio_utils
        juce::juce_dsp
        juce::juce_gui_basics
        juce::juce_gui_extra
        juce::juce_audio_formats
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_lto_flags
        juce::juce_recommended_warning_flags
)

# Custom standalone app: the shared "Close mu-Tilate?" confirmation on quit. Compiled only into the Standalone target.
target_compile_definitions(mu-tilate_Standalone PRIVATE JUCE_USE_CUSTOM_PLUGIN_STANDALONE_APP=1)
target_sources(mu-tilate_Standalone PRIVATE Source/Plugin/StandaloneApp.cpp)

# Hardware controller support (mu-control) in this Standalone only; a no-op unless MUFAMILY_WITH_CONTROL is ON.
mu_control_attach(mu-tilate_Standalone)
mu_asio_attach(mu-tilate_Standalone)   # ASIO devices when -DMUFAMILY_WITH_ASIO=ON

# CLAP format via clap-juce-extensions (mirrors the other products). Creates mu-tilate_CLAP.
clap_juce_extensions_plugin(TARGET mu-tilate
    CLAP_ID "com.transwarp.mu-tilate"
    CLAP_FEATURES instrument sampler)

# Family build-counter (shared increment target defined in mu-clid/CMakeLists.txt).
add_dependencies(mu-tilate increment_build_number)

# Custom Windows VERSIONINFO with the full four-part family version, regenerated by the increment target each build.
if(WIN32)
    configure_file(
        "${CMAKE_CURRENT_SOURCE_DIR}/resources/versioninfo.rc.in"
        "${CMAKE_CURRENT_BINARY_DIR}/versioninfo.rc"
        @ONLY
        NEWLINE_STYLE CRLF   # RC compiler expects CRLF on Windows
    )
    target_compile_definitions(mu-tilate        PRIVATE "JUCE_USER_DEFINED_RC_FILE=\"${CMAKE_CURRENT_BINARY_DIR}/versioninfo.rc\"")
    target_compile_definitions(mu-tilate_rc_lib PRIVATE "JUCE_USER_DEFINED_RC_FILE=\"${CMAKE_CURRENT_BINARY_DIR}/versioninfo.rc\"")
    add_dependencies(mu-tilate_rc_lib increment_build_number)
endif()

# ── mu-tilate-tests — console UnitTests ─────────────────────────────────────
include("${CMAKE_SOURCE_DIR}/mu-core/cmake/ProcessorTestSources.cmake")
juce_add_console_app(mu-tilate-tests PRODUCT_NAME "mu-tilate-tests")

target_sources(mu-tilate-tests PRIVATE
    Source/Tests/TestMain.cpp
    Source/Tests/SlicePlayerTests.cpp   # the monophonic slice player: native / re-pitch / reverse, fades, the cut fade
    Source/Tests/SliceShapingTests.cpp   # per-slice pan, pitch, filter, amp envelope, loop and the pre-fade voice
    Source/Tests/SliceDetectionTests.cpp   # onset detection on synthetic loops, and the slice list
    Source/Tests/StretchSliceTests.cpp   # the stretched slice: length, dry head, pitch and level, no pre-echo, no allocation
    Source/Tests/PhraseTests.cpp   # the timeline model and the transport-anchored scheduler
    Source/Tests/PhraseParamsTests.cpp   # the parameter table and the modulation target table
    Source/Tests/PhraseLayerTests.cpp   # the Phrase layer through the shared render contract
    Source/Tests/PhraseStateTests.cpp   # saving and loading the slices, timeline and loop reference
    Source/Tests/LengthTextTests.cpp   # the timeline length box: what it accepts and shows
    Source/Tests/PhraseLoadTests.cpp   # loading a loop file (cap, errors) and detecting slices on it
    Source/Tests/ProcessorTests.cpp   # the real PluginProcessor, headless: load, play, restore
    # The real PluginProcessor, built headless (createEditor is defined in ProcessorTests.cpp; PluginEditor.cpp is not linked).
    Source/Plugin/PluginProcessor.cpp
    Source/Plugin/PluginProcessor_Preset.cpp
    # The mu-core .cpps behind ProcessorBase (the console target cannot link mu-core directly).
    ${MU_CORE_PROCESSOR_TEST_SOURCES}
    # mu-core's detector (the plugin targets get it through mu-core; a console target lists it).
    "${CMAKE_SOURCE_DIR}/mu-core/Analysis/OnsetDetector.cpp"
    "${CMAKE_SOURCE_DIR}/mu-core/Audio/LoopLoader.cpp"
    # What the modulation-table checks need (as mu-toni-layer-tests lists them).
    "${CMAKE_SOURCE_DIR}/mu-core/Modulation/ModulationMatrix.cpp"
    "${CMAKE_SOURCE_DIR}/mu-core/Sequencer/ControlSequence.cpp"
    # The Layer base and the insert the Phrase layer renders through.
    "${CMAKE_SOURCE_DIR}/mu-core/Audio/InsertProcessor.cpp"
    "${CMAKE_SOURCE_DIR}/mu-core/Audio/MultiModeFilter.cpp"
)
target_include_directories(mu-tilate-tests PRIVATE "${CMAKE_SOURCE_DIR}/mu-core" "${CMAKE_SOURCE_DIR}/mu-core/Tests" "${CMAKE_CURRENT_SOURCE_DIR}/Source")
target_include_directories(mu-tilate-tests SYSTEM PRIVATE "${CMAKE_SOURCE_DIR}/ThirdParty/signalsmith-basics/include" "${CMAKE_SOURCE_DIR}/ThirdParty/signalsmith-dsp/include")   # Tests: AllocationProbe.h

target_compile_features(mu-tilate-tests PRIVATE cxx_std_17)

target_link_libraries(mu-tilate-tests PRIVATE
    mu-tilate-layer
    juce::juce_core
    juce::juce_audio_basics
    juce::juce_audio_formats   # the loop loader
    juce::juce_audio_processors   # the parameter table tests build an APVTS
    juce::juce_data_structures
    juce::juce_events
    juce::juce_graphics
    juce::juce_gui_basics
    juce::juce_dsp   # FFT for the detector
    juce::juce_recommended_config_flags
    juce::juce_recommended_warning_flags
)

target_compile_definitions(mu-tilate-tests PRIVATE JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0 JUCE_USE_MP3AUDIOFORMAT=1
    JUCE_MODAL_LOOPS_PERMITTED=1)   # the processor tests pump the message loop while a loop loads
