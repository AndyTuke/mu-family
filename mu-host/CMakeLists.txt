# mu-Host — one mu instrument channel as a plug-in (VST3 + CLAP, no Standalone).
#
# A product of its own on the mu-Tronic shell: it compiles mu-Tronic's sources (the 16-slot shell, built here with ONE slot:
# MU_MAX_LAYERS=1, no buses) plus its own Source/ (the processor and editor that name it, the engine drop-down, the per-engine
# licensing). Nothing of mu-Tronic is copied; mu-Tronic's own files are shared as they are. See CLAUDE.md.

# The shell's sources, as mu-tronic's target lists them (relative paths made absolute), without its plug-in factory: this product
# has its own.
get_target_property(MU_HOST_SHELL_RAW mu-tronic SOURCES)
set(MU_HOST_SHELL_SOURCES "")
foreach(src ${MU_HOST_SHELL_RAW})
    if(NOT src MATCHES "PluginFactory\\.cpp$")
        get_filename_component(abs "${src}" ABSOLUTE BASE_DIR "${CMAKE_SOURCE_DIR}/mu-tronic")
        list(APPEND MU_HOST_SHELL_SOURCES "${abs}")
    endif()
endforeach()

set(MU_HOST_OWN_SOURCES
    Source/Plugin/PluginProcessor.cpp
    Source/License/EngineLicensing.cpp
)

# ── mu-Host — VST3 + CLAP (+ AU on macOS) ───────────────────────────────────
juce_add_plugin(mu-host
    COMPANY_NAME "Transwarp Development Project"
    BUNDLE_ID "com.transwarp.mu-host"
    PLUGIN_MANUFACTURER_CODE TDP1
    PLUGIN_CODE MHst
    FORMATS VST3 ${MUFAMILY_AU_FORMAT}   # AU on macOS only (family rule); no Standalone
    PRODUCT_NAME "mu-Host"
    PLUGIN_NAME  "μ-Host"
    IS_SYNTH TRUE
    NEEDS_MIDI_INPUT TRUE
    NEEDS_MIDI_OUTPUT TRUE
    IS_MIDI_EFFECT FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS FALSE
    VERSION         "${MUFAMILY_PRE_PLUGIN_VERSION}"
    COMPANY_COPYRIGHT "${MUFAMILY_COPYRIGHT}"
    ICON_BIG   "${CMAKE_SOURCE_DIR}/mu-tronic/resources/icon.png"   # no icon of its own yet
    ICON_SMALL "${CMAKE_SOURCE_DIR}/mu-tronic/resources/icon.png"
)

# Custom VERSIONINFO with the Greek mu (Windows only), regenerated by the increment target each build.
if(WIN32)
    configure_file(
        "${CMAKE_CURRENT_SOURCE_DIR}/resources/versioninfo.rc.in"
        "${CMAKE_CURRENT_BINARY_DIR}/versioninfo.rc"
        @ONLY
        NEWLINE_STYLE CRLF
    )
    target_compile_definitions(mu-host        PRIVATE "JUCE_USER_DEFINED_RC_FILE=\"${CMAKE_CURRENT_BINARY_DIR}/versioninfo.rc\"")
    target_compile_definitions(mu-host_rc_lib PRIVATE "JUCE_USER_DEFINED_RC_FILE=\"${CMAKE_CURRENT_BINARY_DIR}/versioninfo.rc\"")
    add_dependencies(mu-host_rc_lib increment_build_number)
endif()

target_sources(mu-host PRIVATE
    ${MU_HOST_SHELL_SOURCES}
    ${MU_HOST_OWN_SOURCES}
    Source/Plugin/PluginEditor.cpp
)

# mu-Tronic's Source/ comes first, so the shell's own "Plugin/PluginProcessor.h" is the shell's; this product's headers of the same
# names are reached from its own files (the same folder) and by relative path. The repository root is on the path for each product's
# License/LicenseKey.h, which every product names the same.
target_include_directories(mu-host
    PRIVATE
        "${CMAKE_SOURCE_DIR}/mu-tronic/Source"
        "${CMAKE_CURRENT_SOURCE_DIR}/Source"
        "${CMAKE_SOURCE_DIR}/mu-core"
        "${CMAKE_SOURCE_DIR}"
    SYSTEM PRIVATE
        "${CMAKE_SOURCE_DIR}/ThirdParty/signalsmith-basics/include"
        "${CMAKE_SOURCE_DIR}/ThirdParty/signalsmith-dsp/include"
        "${CMAKE_SOURCE_DIR}/ThirdParty/monocypher"
)

target_compile_definitions(mu-host PUBLIC
    MU_TRONIC_WITH_UI=1
    JUCE_WEB_BROWSER=0 JUCE_USE_MP3AUDIOFORMAT=1 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0
)

# The compiled mu-core library for this product's settings (see mu_core_variant in mu-core/CMakeLists.txt).
mu_core_variant(MU_CORE_LIB LAYERS 1 BUSES 0 EXT_SIDECHAIN 0 VERSION_BASE "${MUFAMILY_PRE_VERSION_BASE}")
target_link_libraries(mu-host
    PRIVATE
        mu-toni-layer mu-on-layer mu-tant-layer mu-clid-layer mu-tilate-layer
        ${MU_CORE_LIB}
        juce::juce_audio_utils
        juce::juce_dsp
        juce::juce_gui_basics
        juce::juce_gui_extra
        juce::juce_audio_formats
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_lto_flags
        juce::juce_recommended_warning_flags
)

# Family build counter (the shared increment target is defined in mu-clid/CMakeLists.txt).
add_dependencies(mu-host increment_build_number)

clap_juce_extensions_plugin(TARGET mu-host
    CLAP_ID "com.transwarp.mu-host"
    CLAP_FEATURES instrument synthesizer)

# ── mu-host-tests — the real one-slot processor, headless ───────────────────

# processor, licensing and tests.

set(MU_HOST_TEST_SOURCES "")

        get_filename_component(abs "${src}" ABSOLUTE BASE_DIR "${CMAKE_SOURCE_DIR}/mu-tronic")
        list(APPEND MU_HOST_TEST_SOURCES "${abs}")
    endif()
endforeach()
juce_add_console_app(mu-host-tests PRODUCT_NAME "mu-host-tests")
target_sources(mu-host-tests PRIVATE ${MU_HOST_TEST_SOURCES} ${MU_HOST_OWN_SOURCES} Source/Tests/HostTests.cpp)
target_compile_features(mu-host-tests PRIVATE cxx_std_17)
target_include_directories(mu-host-tests PRIVATE
    "${CMAKE_SOURCE_DIR}/mu-tronic/Source"
    "${CMAKE_CURRENT_SOURCE_DIR}/Source"
    "${CMAKE_SOURCE_DIR}/mu-core"
    "${CMAKE_SOURCE_DIR}"
    SYSTEM
        "${CMAKE_SOURCE_DIR}/ThirdParty/signalsmith-basics/include"
        "${CMAKE_SOURCE_DIR}/ThirdParty/signalsmith-dsp/include"
        "${CMAKE_SOURCE_DIR}/ThirdParty/monocypher")
target_link_libraries(mu-host-tests PRIVATE mu-core-assets
    mu-toni-layer mu-on-layer mu-tant-layer mu-clid-layer mu-tilate-layer
    juce::juce_core juce::juce_audio_basics juce::juce_audio_processors juce::juce_data_structures juce::juce_events
    juce::juce_graphics juce::juce_gui_basics juce::juce_dsp juce::juce_audio_formats juce::juce_audio_utils
    juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)

    MU_MAX_LAYERS=1 MU_MAX_BUSES=0 JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0 JUCE_MODAL_LOOPS_PERMITTED=1)
set_target_properties(mu-host-tests PROPERTIES EXCLUDE_FROM_ALL TRUE)

# ── mu-host-uisoak — the one-slot editor driven in-process ──────────────────
# As mu-tronic-uisoak does for the 16-slot one (mu-tronic/Source/Tests/UiSoakApp.cpp, with UISOAK_HOST): random clicks, drags, keys
# and engine changes handed to the window's peer from inside the process. On demand. It can also write a PNG of the window:
#   MU_UISOAK_SHOT=out.png mu-host-uisoak.exe 3 1
juce_add_plugin(mu-host-uisoak
    COMPANY_NAME "Transwarp Development Project"
    BUNDLE_ID "com.transwarp.mu-host-uisoak"
    PLUGIN_MANUFACTURER_CODE TDP1
    PLUGIN_CODE MHsk
    FORMATS Standalone
    PRODUCT_NAME "mu-host-uisoak"
    PLUGIN_NAME  "hostsoak"
    IS_SYNTH TRUE
    NEEDS_MIDI_INPUT TRUE
    NEEDS_MIDI_OUTPUT TRUE
    IS_MIDI_EFFECT FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS FALSE
    VERSION "${MUFAMILY_PRE_PLUGIN_VERSION}"
    COMPANY_COPYRIGHT "${MUFAMILY_COPYRIGHT}"
)
target_sources(mu-host-uisoak PRIVATE ${MU_HOST_SHELL_SOURCES} ${MU_HOST_OWN_SOURCES} Source/Plugin/PluginEditor.cpp)
target_include_directories(mu-host-uisoak
    PRIVATE
        "${CMAKE_SOURCE_DIR}/mu-tronic/Source"
        "${CMAKE_CURRENT_SOURCE_DIR}/Source"
        "${CMAKE_SOURCE_DIR}/mu-core"
        "${CMAKE_SOURCE_DIR}"
    SYSTEM PRIVATE
        "${CMAKE_SOURCE_DIR}/ThirdParty/signalsmith-basics/include"
        "${CMAKE_SOURCE_DIR}/ThirdParty/signalsmith-dsp/include"
        "${CMAKE_SOURCE_DIR}/ThirdParty/monocypher")
target_compile_definitions(mu-host-uisoak PUBLIC
    MU_MAX_LAYERS=1 MU_MAX_BUSES=0 MU_EXT_SIDECHAIN=0 MU_TRONIC_WITH_UI=1 UISOAK_HOST=1
    JUCE_WEB_BROWSER=0 JUCE_USE_MP3AUDIOFORMAT=1 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0
    "MU_VERSION_BASE=\"${MUFAMILY_PRE_VERSION_BASE}\"")
target_link_libraries(mu-host-uisoak
    PRIVATE mu-toni-layer mu-on-layer mu-tant-layer mu-clid-layer mu-tilate-layer mu-core
        juce::juce_audio_utils juce::juce_dsp juce::juce_gui_basics juce::juce_gui_extra juce::juce_audio_formats
    PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
target_compile_definitions(mu-host-uisoak_Standalone PRIVATE JUCE_USE_CUSTOM_PLUGIN_STANDALONE_APP=1)
target_sources(mu-host-uisoak_Standalone PRIVATE "${CMAKE_SOURCE_DIR}/mu-tronic/Source/Tests/UiSoakApp.cpp")
set_target_properties(mu-host-uisoak mu-host-uisoak_Standalone PROPERTIES EXCLUDE_FROM_ALL TRUE)
